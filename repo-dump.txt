# Clinical benchmark â€” SHA ACLS Provider Manual 2024

Every clinical value in ResusCode, mapped to the Saudi Heart Association (SHA) ACLS
Provider Manual 2024. Where the manual is silent, the AHA 2020/2023 / ERC 2021 value is
kept and marked as such. Code locations are given so each value can be audited.

## CPR & ventilation

| Value in app | SHA 2024 | Code |
|---|---|---|
| Compressions 100â€“120/min, depth 5â€“6 cm, full recoil | Ch. 2 | `engine/store.ts` (start event) |
| 30 : 2 without advanced airway | Ch. 2, Ch. 4 | `hud/InterventionGrid.tsx` banner |
| Advanced airway: continuous compressions + 1 breath every 6 s | Ch. 4 | `hud/InterventionGrid.tsx` banner |
| Switch compressors every 2 min (5 cycles) | Ch. 2, Ch. 8 | `hud/CyclePanel.tsx` |
| Metronome 110 bpm (adjustable 100â€“110) | Ch. 2 (100â€“120/min) | `hud/MetronomeCard.tsx` |
| Airway confirmation with waveform capnography | Ch. 4 | `content/causes.ts` (Hypoxia) |

| Value in app | SHA 2024 | Code |

| Biphasic 120 / 150 / 200 J, monophasic 360 J | Ch. 5: "follow the defibrillator manufacturer for initial and subsequent doses" â€” energies are AHA defaults | `content/calc.ts` `SHOCK_ENERGIES` |

## Reversible causes (Table 3.4)

SHA lists **six H's** â€” Hypoxia, Hypovolaemia, Hypothermia, Hydrogen ion, Hyperkalaemia
and **Hypoglycaemia** â€” and the T's Thrombosis (pulmonary), Thrombosis (coronary),
Tamponade and Toxins. Tension pneumothorax is retained from AHA/ERC. Clues and
countermeasures in `content/causes.ts` follow the table's risk factors and management
columns (e.g. hyperkalaemia: calcium chloride, glucose + insulin, sodium bicarbonate,
nebulised albuterol, diuretics, dialysis; PE: empirical thrombolysis when suspected,
embolectomy, eCPR â€” Ch. 10 Special circumstances).

import { useEffect, useRef } from 'react';
import { useShallow } from 'zustand/react/shallow';
import { RotateCcw, Timer, Waves } from 'lucide-react';
import { useCodeStore } from '../../engine/store';
import { useNow } from '../../hooks/useNow';
import {
  CRIT_MS,
  CYCLE_MS,
  cyclePhase,
  cycleProgress,
  cycleRemainingMs,
  formatCountdown,
} from '../../engine/timing';
import { playKlaxon, playWarningChimes } from '../../engine/audio';
import { haptic } from '../../engine/haptics';
import type { RhythmLoop } from '../../engine/types';

const R = 108;
const C = 2 * Math.PI * R;

const PHASE_COLOR: Record<string, string> = {
  cruise: '#06b6d4',
  warn: '#f59e0b',
  crit: '#ef4444',
};

const RHYTHMS: RhythmLoop[] = ['VF', 'pVT', 'PEA', 'Asystole', 'Pulse'];

/**
 * The 2-minute CPR cycle. Cyan â†’ amber (â‰¤ 30 s) â†’ flashing crimson (â‰¤ 10 s, with a
 * 3-chime warning), then a klaxon and a giant "RHYTHM CHECK / SWITCH COMPRESSORS"
 * alert at 00:00. The cycle timer is independent of the master clock, so resetting
 * it never disturbs the arrest duration.
 */
export function CyclePanel() {
  const now = useNow();
  const cyclic = useCodeStore(
    useShallow((s) => ({
      status: s.status,
      cycleIndex: s.cycleIndex,
      startedAt: s.startedAt,
      pausedAt: s.pausedAt,
      pausedAccumMs: s.pausedAccumMs,
      cycleStartedAt: s.cycleStartedAt,
      cyclePausedAccumMs: s.cyclePausedAccumMs,
    })),
  );
  const resetCycle = useCodeStore((s) => s.resetCycle);

  const remaining = cycleRemainingMs(cyclic, now);
  const phase = cyclePhase(remaining);
  const progress = cycleProgress(remaining);
  const active = cyclic.status === 'active';
  const expired = active && remaining <= 0;
  const warning = active && !expired && remaining <= CRIT_MS;

  // Each alert fires exactly once per cycle, and re-arms when the cycle is reset.
  const warned = useRef(false);
  const fired = useRef(false);
  useEffect(() => {
    if (warning && !warned.current) {
      warned.current = true;
      playWarningChimes();
      haptic.warn();
    }
    if (!warning && !expired) warned.current = false;
  }, [warning, expired]);
  useEffect(() => {
    if (expired && !fired.current) {
      fired.current = true;
      playKlaxon();
      haptic.alarm();
    }
    if (!expired) fired.current = false;
  }, [expired]);

  const color = PHASE_COLOR[phase];
  const arcLen = C * (1 - progress);

  const label = !active
    ? cyclic.status === 'idle'
      ? 'Awaiting start'
      : 'Cycle stopped'
    : expired
      ? 'Cycle complete'
      : 'Cycle remaining';

  return (
    <div
      className={`rounded-2xl border bg-panel/80 p-3 ${
        expired ? 'border-crit' : phase === 'crit' && active ? 'border-crit/70' : phase === 'warn' ? 'border-warn/50' : 'border-line'
      }`}
    >
      <div className="mb-1 flex items-center justify-between">
        <span className="flex items-center gap-1.5 text-xs font-bold uppercase tracking-widest text-mute">
          <Timer className="h-4 w-4" /> 2-min CPR cycle
        </span>
        <span className="tnum rounded-full border border-line px-2 py-0.5 text-xs font-bold text-mute">
          Cycle {cyclic.cycleIndex}
        </span>
      </div>

      {expired ? (
        <div className="animate-flash-bg mb-2 rounded-2xl border-2 border-crit px-3 py-4 text-center text-white">
          <p className="text-2xl font-black leading-tight tracking-tight sm:text-3xl">RHYTHM CHECK</p>
          <p className="text-xl font-black leading-tight tracking-tight sm:text-2xl">SWITCH COMPRESSORS</p>
        </div>
      ) : null}

      <div className="relative mx-auto flex h-[236px] w-[236px] items-center justify-center">
        <svg viewBox="0 0 260 260" className="absolute inset-0 h-full w-full -rotate-90">
          <circle cx="130" cy="130" r={R} fill="none" stroke="#131c2c" strokeWidth="18" />
          <circle
            cx="130"
            cy="130"
            r={R}
            fill="none"
            stroke={color}
            strokeWidth="18"
            strokeLinecap="round"
            strokeDasharray={`${arcLen} ${C}`}
            style={{ transition: 'stroke 300ms linear' }}
          />
        </svg>
        <div className="flex flex-col items-center">
          <span
            className={`tnum font-mono text-6xl font-bold leading-none ${
              expired || warning ? 'animate-flash text-crit' : ''
            }`}
            style={{ color: expired || warning ? undefined : color }}
          >
            {formatCountdown(remaining)}
          </span>
          <span
            className={`mt-1 max-w-[170px] text-center text-[11px] font-bold uppercase tracking-wider ${
              expired ? 'animate-flash text-crit' : 'text-mute'
            }`}
          >
            {label}
          </span>
        </div>
      </div>

      {expired ? (
        <div className="mb-2 rounded-xl border border-crit p-2">
          <p className="mb-2 flex items-center justify-center gap-2 text-center text-sm font-bold text-crit">
            <Waves className="h-4 w-4" /> Log the rhythm â€” resets the cycle
          </p>
          <div className="grid grid-cols-5 gap-1.5">
            {RHYTHMS.map((r) => (
              <button
                key={r}
                type="button"
                onClick={() => resetCycle(r)}
                className="touch-hit min-h-[64px] rounded-xl border border-crit/60 bg-crit/10 text-sm font-bold text-ink active:scale-[0.96]"
              >
                {r}
              </button>
            ))}
          </div>
        </div>
      ) : null}

      <button
        type="button"
        onClick={() => resetCycle()}
        disabled={!active}
        className={[
          'touch-hit tnum flex min-h-[72px] w-full items-center justify-center gap-2 rounded-2xl border text-lg font-bold active:scale-[0.98]',
          phase === 'crit' && active
            ? 'animate-flash-bg border-crit text-white'
            : 'border-cyber/60 bg-cyber/15 text-cyber',
          active ? '' : 'pointer-events-none opacity-40',
        ].join(' ')}
      >
        <RotateCcw className="h-6 w-6" />
        Reset 2-Min Cycle
      </button>
      <p className="mt-1 text-center text-[11px] text-mute">
        Total arrest clock keeps running Â· cycle length {CYCLE_MS / 1000}s
      </p>
    </div>
  );
}

import { ClipboardCopy, Download, Eye, RotateCcw, Check } from 'lucide-react';

/** SBAR debrief export â€” copy to the EMR clipboard or download a .txt. */
export function SbarExport() {
  const eventCount = useCodeStore((s) => s.events.length);
  const reset = useCodeStore((s) => s.reset);
  const [copied, setCopied] = useState(false);
  const [preview, setPreview] = useState<string | null>(null);

  const build = () => buildSbar(useCodeStore.getState(), Date.now());

  const copy = async () => {
    const text = build();
    try {
      if (navigator.clipboard?.writeText) {
        await navigator.clipboard.writeText(text);
      } else {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        document.execCommand('copy');
        document.body.removeChild(ta);
      }
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    } catch {
      setPreview(text);
    }
  };

  const download = () => {
    const text = build();
    const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = sbarFilename(Date.now());
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

  return (
    <Panel title="SBAR export" icon={ClipboardCopy} accent="cyber">
      <BigButton
        variant={copied ? 'good' : 'cyber'}
        size="xl"
        icon={copied ? Check : ClipboardCopy}
        label={copied ? 'Copied to clipboard' : 'Copy SBAR Resuscitation Summary'}
        sub="Plain-text note ready to paste into the EHR/EMR"
        disabled={eventCount === 0}
        onClick={copy}
      />
      <div className="mt-2 grid grid-cols-2 gap-2">
        <BigButton
          variant="neutral"
          icon={Download}
          label="Download .txt"
          disabled={eventCount === 0}
          onClick={download}
        />
        <BigButton
          variant="outline"
          icon={Eye}
          label={preview ? 'Hide preview' : 'Preview'}
          disabled={eventCount === 0}
          onClick={() => setPreview((p) => (p ? null : build()))}
        />
      </div>

      {preview ? (
        <pre className="selectable mt-2 max-h-72 overflow-auto whitespace-pre-wrap rounded-xl border border-line bg-black/40 p-3 font-mono text-[11px] leading-snug text-ink">
          {preview}

/**
 * Checked state derived from the persisted event log, so ticks survive tab switches
 * and refreshes. Unticking logs an explicit "unchecked" entry rather than erasing history.
 */
export function useLoggedChecklist(category: EventCategory, prefix: string) {
  const events = useCodeStore((s) => s.events);
  const logEvent = useCodeStore((s) => s.logEvent);

  const onLabel = (text: string) => `${prefix}: ${text}`;
  const offLabel = (text: string) => `${prefix} (unchecked): ${text}`;

  const isOn = (text: string) => {
    let state = false;
    for (const e of events) {
      if (e.label === onLabel(text)) state = true;
      else if (e.label === offLabel(text)) state = false;
    }
    return state;
  };

  const toggle = (text: string) => logEvent(category, isOn(text) ? offLabel(text) : onLabel(text));

  return { isOn, toggle };
}

  primary: { base: 'border-cyber/60 text-cyber', fill: 'bg-cyber/30' },

export interface HoldButtonProps {
  label: string;
  sub?: string;
  icon?: LucideIcon;
  onComplete: () => void;
  /** hold duration in ms before the action fires (default 1.5 s) */
  durationMs?: number;
  variant?: ButtonVariant;
  size?: 'lg' | 'xl';
  className?: string;
  disabled?: boolean;
  ariaLabel?: string;

/**
 * Press-and-hold to confirm â€” a deliberate gate against accidental taps on
 * irreversible actions (ROSC / terminate) WITHOUT the friction of a modal dialog.
 * A fill sweeps across the button so the user always sees how much longer to hold.
 */
export function HoldButton({
  label,
  sub,
  icon: Icon,
  onComplete,
  durationMs = 1500,
  variant = 'neutral',
  size = 'lg',
  className = '',
  disabled,
  ariaLabel,
}: HoldButtonProps) {
  const [pct, setPct] = useState(0);
  const raf = useRef<number | null>(null);
  const startedAt = useRef(0);
  const fired = useRef(false);

  const cancel = useCallback(() => {
    if (raf.current != null) {
      cancelAnimationFrame(raf.current);
      raf.current = null;
    }
    if (!fired.current) setPct(0);
  }, []);

  const loop = useCallback(() => {
    const elapsed = performance.now() - startedAt.current;
    const p = Math.min(1, elapsed / durationMs);
    setPct(p);
    if (p >= 1) {
      fired.current = true;
      if (raf.current != null) cancelAnimationFrame(raf.current);
      raf.current = null;
      haptic.alarm();
      onComplete();
      // release visual after a beat
      setTimeout(() => {
        fired.current = false;
        setPct(0);
      }, 250);
      return;
    }
    raf.current = requestAnimationFrame(loop);
  }, [durationMs, onComplete]);

  const begin = useCallback(() => {
    if (disabled) return;
    startedAt.current = performance.now();
    fired.current = false;
    raf.current = requestAnimationFrame(loop);
  }, [disabled, loop]);

  const skin = VARIANT_SKIN[variant];
  const height = size === 'xl' ? 'min-h-[84px]' : 'min-h-[64px]';

  return (
    <button
      type="button"
      aria-label={ariaLabel ?? `${label} â€” press and hold to confirm`}
      disabled={disabled}
      onPointerDown={(e) => {
        e.currentTarget.setPointerCapture?.(e.pointerId);
        begin();
      }}
      onPointerUp={cancel}
      onPointerCancel={cancel}
      onPointerLeave={cancel}
      onContextMenu={(e) => e.preventDefault()}
      className={[
        'touch-hit relative flex w-full items-center gap-3 overflow-hidden rounded-2xl border px-4 text-left',
        'transition-[transform] duration-100 active:scale-[0.98]',
        height,
        skin.base,
        disabled ? 'pointer-events-none opacity-40' : '',
        className,
      ].join(' ')}
    >
      <span
        aria-hidden
        className={`absolute inset-y-0 left-0 ${skin.fill} transition-none`}
        style={{ width: `${pct * 100}%` }}
      />
      <span className="relative flex w-full items-center gap-3">
        {Icon ? <Icon className="h-7 w-7 shrink-0" strokeWidth={2.4} aria-hidden /> : null}
        <span className="flex min-w-0 flex-col">

/**
 * Audio engine â€” 100% synthesized via the Web Audio API.
 *
 * No external .mp3/.wav assets exist anywhere in this app, so audio cannot fail
 * to load offline. The AudioContext is created lazily on the first user gesture
 * to satisfy mobile autoplay policies.
 */

let ctx: AudioContext | null = null;

type AudioCtor = typeof AudioContext;

function getAudioCtor(): AudioCtor | null {
  if (typeof window === 'undefined') return null;
  const w = window as unknown as { AudioContext?: AudioCtor; webkitAudioContext?: AudioCtor };
  return w.AudioContext ?? w.webkitAudioContext ?? null;
}

/** Create/resume the shared AudioContext. Safe to call repeatedly. Returns null when unsupported. */
export function ensureAudio(): AudioContext | null {
  const Ctor = getAudioCtor();
  if (!Ctor) return null;
  if (!ctx) {
    try {
      ctx = new Ctor();
    } catch {
      return null;
    }
  }
  if (ctx.state === 'suspended') void ctx.resume().catch(() => {});
  return ctx;
}

export function isAudioReady(): boolean {
  return ctx != null && ctx.state === 'running';
}

interface ClickOptions {
  freq: number;
  type?: OscillatorType;
  gain?: number;
  durationMs?: number;
}

/** Schedule a single enveloped tone at an absolute AudioContext time. */
export function scheduleClick(when: number, opts: ClickOptions): void {
  const ac = ensureAudio();
  if (!ac) return;
  const t = Math.max(when, ac.currentTime);
  const dur = (opts.durationMs ?? 45) / 1000;
  const peak = opts.gain ?? 0.16;

  const osc = ac.createOscillator();
  const g = ac.createGain();
  osc.type = opts.type ?? 'square';
  osc.frequency.setValueAtTime(opts.freq, t);

  g.gain.setValueAtTime(0.0001, t);
  g.gain.linearRampToValueAtTime(peak, t + 0.004);
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur);

  osc.connect(g).connect(ac.destination);
  osc.start(t);
  osc.stop(t + dur + 0.03);
}

  osc.frequency.setValueAtTime(from, when);

/**
 * Hybrid precision clock.
 *
 * Absolute epoch timestamps remain the persisted source of truth (they survive a
 * refresh, which `performance.now()` does not). Live time, however, advances on the
 * monotonic `performance.now()` delta from an epoch anchor, so a wall-clock change
 * (NTP sync, manual time change) cannot make the running timers jump.
 *
 * Some platforms pause `performance.now()` while the device sleeps. We re-anchor when
 * the page becomes visible again, and whenever the monotonic estimate drifts from
 * `Date.now()` by more than RESYNC_MS (which only happens after a suspend).
 */

export const RESYNC_MS = 1_000;

interface PerfLike {
  now(): number;
}

function perf(): PerfLike | null {
  return typeof performance !== 'undefined' && typeof performance.now === 'function' ? performance : null;
}

let anchorEpoch = Date.now();
let anchorPerf = perf()?.now() ?? 0;

/** Re-anchor the monotonic clock to the current wall clock. */
export function resyncClock(): void {
  anchorEpoch = Date.now();
  anchorPerf = perf()?.now() ?? 0;
}

/** Epoch ms advanced by the monotonic clock. Use everywhere instead of Date.now(). */
export function clockNow(): number {
  const p = perf();
  if (!p) return Date.now();
  const mono = Math.round(anchorEpoch + (p.now() - anchorPerf));
  const wall = Date.now();
  if (Math.abs(mono - wall) > RESYNC_MS) {
    // Suspend/resume or a large clock correction: trust the wall clock and re-anchor.
    resyncClock();
    return wall;
  }
  return mono;
}

  }
}

const V2_CAUSES: ReadonlySet<string> = new Set<CauseId>([
  'hypovolemia',
  'hypoxia',
  'hydrogen_ion',
  'hypo_hyperkalemia',
  'hypothermia',
  'hypoglycemia',
  'tension_ptx',

/**
 * Upgrade a CodeOR v1 snapshot. Perioperative crisis fields are dropped, the single
 * `thrombosis` cause becomes `thrombosis_pulmonary`, and new fields get defaults.
 * Crisis events already in the log are kept (re-filed as `system`) so the audit
 * trail is never rewritten.
 */
export function migrateV1(raw: unknown): PersistedSession | null {
  if (!raw || typeof raw !== 'object') return null;
  const v1 = raw as Record<string, unknown>;
  if (v1.version !== 1 || !Array.isArray(v1.events)) return null;

  const base = emptySession();
  const causes: Partial<Record<CauseId, boolean>> = {};
  const oldCauses = (v1.causes ?? {}) as Record<string, boolean>;
  for (const [k, on] of Object.entries(oldCauses)) {
    const id = k === 'thrombosis' ? 'thrombosis_pulmonary' : k;
    if (V2_CAUSES.has(id)) causes[id as CauseId] = !!on;
  }

  const events = (v1.events as PersistedSession['events']).map((e) =>
    (e.category as string) === 'crisis' ? { ...e, category: 'system' as const } : e,
  );

  return {
    ...base,
    status: (v1.status as PersistedSession['status']) ?? base.status,
    startedAt: (v1.startedAt as number | null) ?? null,
    pausedAt: (v1.pausedAt as number | null) ?? null,
    pausedAccumMs: (v1.pausedAccumMs as number) ?? 0,
    cycleStartedAt: (v1.cycleStartedAt as number | null) ?? null,
    cyclePausedAccumMs: (v1.cyclePausedAccumMs as number) ?? 0,
    cycleIndex: (v1.cycleIndex as number) ?? 1,
    events,
    shocks: (v1.shocks as PersistedSession['shocks']) ?? [],
    epi: (v1.epi as PersistedSession['epi']) ?? [],
    amio: (v1.amio as PersistedSession['amio']) ?? [],
    airwayPlaced: !!v1.airwayPlaced,
    airwayType: v1.airwayPlaced ? 'ETT' : null,
    causes,
    metronome: { ...base.metronome, ...((v1.metronome as object) ?? {}) },
    weightKg: (v1.weightKg as number) ?? DEFAULT_WEIGHT_KG,
    settings: { ...base.settings, ...((v1.settings as object) ?? {}) },
    savedAt: (v1.savedAt as number) ?? 0,
  };
}

function isBrowser(): boolean {
  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined';
}

export function loadPersisted(): PersistedSession | null {
  if (!isBrowser()) return null;
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);
    if (raw) {
      const parsed = JSON.parse(raw) as PersistedSession;
      if (!parsed || parsed.version !== 2 || !Array.isArray(parsed.events)) return null;
      return { ...emptySession(), ...parsed };
    }
    const legacy = window.localStorage.getItem(LEGACY_STORAGE_KEY);
    if (!legacy) return null;
    const migrated = migrateV1(JSON.parse(legacy));
    window.localStorage.removeItem(LEGACY_STORAGE_KEY);
    if (migrated) writePersisted(migrated);
    return migrated;
  } catch {
    return null;
  }
}

export function writePersisted(session: PersistedSession): void {
  if (!isBrowser()) return;
  try {
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(session));
  } catch {
    /* quota or private-mode failure â€” the in-memory session still works */
  }
}

export function clearPersisted(): void {
  if (!isBrowser()) return;
  try {
    window.localStorage.removeItem(STORAGE_KEY);
  } catch {
    /* ignore */
  }
}

/**
 * Debounced writer: the store mutates frequently, so we coalesce writes.
 * `flush` is called on pagehide to guarantee the final state is durable.
 */
export function createDebouncedWriter(delayMs = 400) {
  let handle: ReturnType<typeof setTimeout> | null = null;
  let pending: PersistedSession | null = null;

  const flush = () => {
    if (handle != null) {
      clearTimeout(handle);
      handle = null;
    }
    if (pending) {
      writePersisted(pending);
      pending = null;
    }
  };

  const save = (session: PersistedSession) => {
    pending = session;
    if (handle != null) return;
    handle = setTimeout(() => {
      handle = null;
      if (pending) {
        writePersisted(pending);
        pending = null;
      }
    }, delayMs);
  };

  return { save, flush };
}

 * SBAR resuscitation summary â€” the plain-text debrief that gets pasted into the
 * EMR/EHR code documentation note. Pure and unit-tested so the output stays stable.
 *
 * Contains no patient identifiers by design (PHI-free).
 */

import type {
  AirwayType,
  CauseId,
  CodeEvent,
  DoseEvent,
  PocusId,
  RhythmBranch,
  SessionStatus,
  ShockEvent,
} from './types';
import { formatClock, formatDuration, formatWallClock, masterElapsedMs } from './timing';
import { CAUSES } from '../content/causes';
import { POCUS_ITEMS } from '../content/pocus';
import { waveformFor, waveformLabel } from '../content/calc';

export interface SbarInput {
  status: SessionStatus;
  startedAt: number | null;
  pausedAt: number | null;
  pausedAccumMs: number;
  events: CodeEvent[];
  shocks: ShockEvent[];
  epi: DoseEvent[];
  amio: DoseEvent[];
  lido: DoseEvent[];
  rhythmBranch: RhythmBranch | null;
  airwayPlaced: boolean;
  airwayType: AirwayType | null;
  causes: Partial<Record<CauseId, boolean>>;
  pocus: Partial<Record<PocusId, boolean>>;

}

const STATUS_LABEL: Record<SessionStatus, string> = {
  idle: 'Not started',
  active: 'In progress â€” CPR ongoing',
  paused: 'In progress â€” CPR paused',
  rosc: 'ROSC achieved',
  terminated: 'Resuscitation terminated',
};

  shockable: 'Shockable (VF / pulseless VT)',
  nonshockable: 'Non-shockable (Asystole / PEA)',

export function buildSbar(input: SbarInput, now: number = Date.now()): string {

  const arrest = input.startedAt != null;
  const rosc = input.events.find((e) => e.category === 'rosc');
  const addressed = CAUSES.filter((c) => input.causes[c.id]).map((c) => c.label);
  const pocus = POCUS_ITEMS.filter((p) => input.pocus[p.id]).map((p) => p.label);
  const epiMg = Math.round(input.epi.reduce((s, d) => s + (d.mg ?? 0), 0) * 100) / 100;
  const cycles = input.events.filter((e) => e.category === 'cycle').length;
  const rhythms = input.events
    .filter((e) => e.category === 'rhythm' && e.label.startsWith('Rhythm check â€” '))
    .map((e) => e.label.replace('Rhythm check â€” ', ''));

  const out: string[] = [];

  out.push('ResusCode â€” RESUSCITATION SUMMARY');
  out.push(`Generated: ${new Date(now).toLocaleString()}`);
  if (arrest) {
    out.push(`Total arrest time: ${formatClock(total)} (${formatDuration(total)})`);
    out.push(`Outcome: ${STATUS_LABEL[input.status]}`);
  } else {
    out.push('Outcome: No cardiac arrest recorded');
  }
  out.push('(PHI-free â€” no patient identifiers recorded)');
  out.push('');

  out.push('=== SITUATION ===');
  if (arrest) {
    out.push('Adult in-hospital cardiac arrest â€” Code Blue resuscitation per SHA ACLS 2024 (AHA/ERC-aligned).');
    out.push(`Code activated at ${formatWallClock(input.startedAt!)}. Arrest duration: ${formatDuration(total)}.`);
    if (rosc) out.push(`ROSC achieved at [${formatClock(rosc.t)}] (${formatWallClock(rosc.at)}).`);
  } else {
    out.push('No cardiac arrest recorded.');
  }
  out.push('');

  out.push('');

    const energies = input.shocks

      .join(', ');
    out.push(`Defibrillation energies: ${energies}`);
  }
  out.push(`Reversible causes addressed: ${addressed.length ? addressed.join(', ') : 'none recorded'}`);
  out.push(`POCUS findings: ${pocus.length ? pocus.join(', ') : 'none recorded'}`);
  out.push('');

  out.push('=== RECOMMENDATIONS ===');
  if (input.status === 'rosc') {
    out.push('- Post-ROSC bundle: secure airway (ETT), SpOâ‚‚ 92â€“98%, avoid hyperventilation.');
    out.push('- Hypotension: IV fluid up to 2 L, then dopamine 5â€“10 mcg/kg/min (or norepinephrine/epinephrine); MAP â‰¥ 65 mmHg.');
    out.push('- 12-lead ECG â€” Cath Lab for STEMI; if unresponsive, temperature control 32â€“37.5 Â°C for 12â€“24 h.');

}

/**
 * Pure time math for ResusCode.
 *
 * Design law: absolute timestamps are the source of truth; the render ticker only
 * repaints. This makes the master clock and the 2-minute cycle resilient to page
 * refreshes, background throttling, and pause/resume without accumulating drift.
 *
 * All functions here are pure and fully unit-tested (see timing.test.ts).
 */

export const CYCLE_MS = 120_000; // 2:00 CPR cycle
export const WARN_MS = 30_000; // cyan -> amber
export const CRIT_MS = 10_000; // amber -> flashing red + alarm

export interface ClockState {
  startedAt: number | null;
  pausedAt: number | null;
  pausedAccumMs: number;
}

export interface CycleClockState extends ClockState {
  cycleStartedAt: number | null;
  cyclePausedAccumMs: number;
}

/** Total elapsed arrest time (ms), pause-aware. */
export function masterElapsedMs(s: ClockState, now: number): number {
  if (s.startedAt == null) return 0;
  const ref = s.pausedAt ?? now;
  return Math.max(0, ref - s.startedAt - s.pausedAccumMs);
}

/** Elapsed time within the current CPR cycle (ms), pause-aware and independent of the master clock. */
export function cycleElapsedMs(s: CycleClockState, now: number): number {
  if (s.cycleStartedAt == null) return CYCLE_MS;
  const ref = s.pausedAt ?? now;
  const elapsed = ref - s.cycleStartedAt - s.cyclePausedAccumMs;
  return Math.max(0, Math.min(CYCLE_MS, elapsed));
}

/** Time remaining in the current CPR cycle (ms), clamped to [0, CYCLE_MS]. */
export function cycleRemainingMs(s: CycleClockState, now: number): number {
  return Math.max(0, CYCLE_MS - cycleElapsedMs(s, now));
}

export type CyclePhase = 'cruise' | 'warn' | 'crit';

export function cyclePhase(remainingMs: number): CyclePhase {
  if (remainingMs <= CRIT_MS) return 'crit';
  if (remainingMs <= WARN_MS) return 'warn';
  return 'cruise';
}

/** Fraction of the cycle completed, 0..1 â€” used to fill the progress ring. */
export function cycleProgress(remainingMs: number): number {
  const done = CYCLE_MS - Math.max(0, Math.min(CYCLE_MS, remainingMs));
  return done / CYCLE_MS;
}

function pad2(n: number): string {
  return n < 10 ? `0${n}` : `${n}`;
}

/**
 * Format ms as MM:SS, promoting to H:MM:SS past one hour.
 * Always returns the same number of digits for a given magnitude to avoid
 * visual jumpiness in the giant monospace readout.
 */
export function formatClock(ms: number): string {
  const total = Math.max(0, Math.floor(ms / 1000));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  if (h > 0) return `${h}:${pad2(m)}:${pad2(s)}`;
  return `${pad2(m)}:${pad2(s)}`;
}

/** Human duration for summaries: "6m 45s", "1h 02m 05s", "45s". */
export function formatDuration(ms: number): string {
  const total = Math.max(0, Math.floor(ms / 1000));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  if (h > 0) return `${h}h ${pad2(m)}m ${pad2(s)}s`;
  if (m > 0) return `${m}m ${pad2(s)}s`;
  return `${s}s`;
}

/** Local time-of-day HH:MM:SS for an absolute epoch timestamp (for documentation). */
export function formatWallClock(epochMs: number): string {
  const d = new Date(epochMs);
  return `${pad2(d.getHours())}:${pad2(d.getMinutes())}:${pad2(d.getSeconds())}`;
}

/** Same as formatClock; kept separate for intent (countdowns) at call sites. */
export function formatCountdown(ms: number): string {
  return formatClock(ms);
}

/** Milliseconds remaining until epinephrine may be re-dosed (0 when eligible). */
export function epiLockoutRemainingMs(
  lastDoseAt: number | null | undefined,
  lockoutMs: number,
  now: number,
): number {
  if (lastDoseAt == null) return 0;
  return Math.max(0, lastDoseAt + lockoutMs - now);
}

/** mm:ss string for a dose lockout that is still counting down, or null when eligible. */
export function epiLockoutLabel(
  lastDoseAt: number | null | undefined,
  lockoutMs: number,
  now: number,
): string | null {
  const remaining = epiLockoutRemainingMs(lastDoseAt, lockoutMs, now);
  if (remaining <= 0) return null;
  return formatCountdown(remaining);
}

/**
 * Wake Lock â€” the screen must never dim or sleep while a code is running.
 *
 * The browser releases the lock whenever the page becomes hidden, so we
 * re-acquire on visibilitychange. All access is guarded so the app behaves on
 * browsers without the API (older Safari) â€” the UI surfaces a fallback badge.
 */

// Minimal structural types so this compiles regardless of the DOM lib version.
interface WakeLockSentinelLike {
  released: boolean;
  release: () => Promise<void>;
  addEventListener: (type: 'release', listener: () => void) => void;
}

interface WakeLockLike {
  request: (type: 'screen') => Promise<WakeLockSentinelLike>;
}

type NavWithWakeLock = Navigator & { wakeLock?: WakeLockLike };

let sentinel: WakeLockSentinelLike | null = null;
let listenerAttached = false;
const subscribers = new Set<(active: boolean) => void>();

export function isWakeLockSupported(): boolean {
  return typeof navigator !== 'undefined' && 'wakeLock' in navigator;
}

export function isWakeLockActive(): boolean {
  return sentinel != null && !sentinel.released;
}

function notify(): void {
  const active = isWakeLockActive();
  subscribers.forEach((fn) => fn(active));
}

export function subscribeWakeLock(fn: (active: boolean) => void): () => void {
  subscribers.add(fn);
  fn(isWakeLockActive());
  return () => subscribers.delete(fn);
}

async function acquire(): Promise<void> {
  const nav = navigator as NavWithWakeLock;
  if (!nav.wakeLock) return;
  try {
    sentinel = await nav.wakeLock.request('screen');
    if (!listenerAttached) {
      listenerAttached = true;
      document.addEventListener('visibilitychange', () => {
        if (document.visibilityState === 'visible' && wantLock) void acquire();
      });
    }
    // Auto re-acquire if the platform drops the lock while we still want it.
    sentinel.addEventListener('release', () => {
      if (wantLock && document.visibilityState === 'visible') void acquire();
      else notify();
    });
    notify();
  } catch {
    notify();
  }
}

let wantLock = false;

export async function requestWakeLock(): Promise<void> {
  wantLock = true;
  if (isWakeLockActive()) return;
  await acquire();
}

export async function releaseWakeLock(): Promise<void> {
  wantLock = false;
  const s = sentinel;
  sentinel = null;
  if (s && !s.released) {
    try {
      await s.release();
    } catch {
      /* ignore */
    }
  }
  notify();
}

import { useEffect, useRef, useState } from 'react';
import { useCodeStore } from '../engine/store';
import { ensureAudio, scheduleClick } from '../engine/audio';
import { haptic } from '../engine/haptics';

const LOOKAHEAD_S = 0.1; // schedule audio this far ahead
const SCHEDULER_MS = 25; // how often the scheduler wakes
const ACCENT_EVERY = 10; // accent the down-beat every 10th tick (~every 6 s)

/**
 * CPR metronome. Uses a lookahead scheduler on a single shared AudioContext so
 * the cadence is rock-steady at 110 bpm (naive setInterval drifts audibly).
 * Returns a beat counter so the visual indicator can pulse in lock-step.
 */
export function useMetronome(): { beat: number; running: boolean } {
  const on = useCodeStore((s) => s.metronome.on);
  const bpm = useCodeStore((s) => s.metronome.bpm);
  const audioOn = useCodeStore((s) => s.metronome.audio);
  const hapticOn = useCodeStore((s) => s.metronome.haptic);
  const status = useCodeStore((s) => s.status);

  const [beat, setBeat] = useState(0);
  // Silent while CPR is paused â€” a ticking metronome during a pause invites resuming early.
  const running = on && status === 'active' && bpm > 0;

  // Keep latest toggles in refs so the scheduler effect does not restart on every change.
  const audioRef = useRef(audioOn);
  const hapticRef = useRef(hapticOn);
  audioRef.current = audioOn;
  hapticRef.current = hapticOn;

  useEffect(() => {
    if (!running) return;
    const ac = ensureAudio();
    const interval = 60 / bpm;
    let stopped = false;
    let next = ac ? ac.currentTime + 0.08 : 0;
    let count = 0;
    const visualTimers: number[] = [];

    const schedule = () => {
      if (!ac || stopped) return;
      while (next < ac.currentTime + LOOKAHEAD_S) {
        const accent = count % ACCENT_EVERY === 0;
        if (audioRef.current) {
          scheduleClick(next, {
            freq: accent ? 1320 : 880,
            type: 'square',
            gain: accent ? 0.2 : 0.13,
            durationMs: 34,
          });
        }
        const delayMs = Math.max(0, (next - ac.currentTime) * 1000);
        const isHapticBeat = hapticRef.current;
        const id = window.setTimeout(() => {
          if (stopped) return;
          setBeat((b) => b + 1);
          if (isHapticBeat) haptic.tick();
        }, delayMs);
        visualTimers.push(id);
        next += interval;
        count += 1;
      }
    };

    const handle = window.setInterval(schedule, SCHEDULER_MS);
    schedule();

    return () => {
      stopped = true;
      window.clearInterval(handle);
      visualTimers.forEach((id) => window.clearTimeout(id));
    };
  }, [running, bpm]);

  return { beat, running };
}

import { useEffect } from 'react';
import { useCodeStore } from '../engine/store';
import { clockNow, resyncClock } from '../engine/clock';

/**
 * Single app-wide render ticker. Updates `now` in the store so every clock
 * re-derives from real timestamps. rAF-based (throttled to ~5 Hz) on the monotonic
 * hybrid clock, and paused entirely when there is no running code (battery).
 */
export function useTicker(): void {
  const status = useCodeStore((s) => s.status);
  const tick = useCodeStore((s) => s.tick);
  const metronomeOn = useCodeStore((s) => s.metronome.on);

  useEffect(() => {
    const live = status === 'active' || status === 'paused';
    tick(clockNow());
    if (!live && !metronomeOn) return;

    let raf = 0;
    let last = 0;
    const loop = (t: number) => {
      if (t - last >= 200) {
        last = t;
        tick(clockNow());
      }
      raf = requestAnimationFrame(loop);
    };
    raf = requestAnimationFrame(loop);

    const onVisible = () => {
      resyncClock();
      last = 0;
      tick(clockNow());
    };
    document.addEventListener('visibilitychange', onVisible);

    return () => {
      cancelAnimationFrame(raf);
      document.removeEventListener('visibilitychange', onVisible);
    };
  }, [status, metronomeOn, tick]);
}

/** Current ticker value. Only components that need live time should use this. */
export function useNow(): number {
  return useCodeStore((s) => s.now);

function useOnline(): boolean {
  const [online, setOnline] = useState(typeof navigator === 'undefined' ? true : navigator.onLine);
  useEffect(() => {
    const on = () => setOnline(true);
    const off = () => setOnline(false);
    window.addEventListener('online', on);
    window.addEventListener('offline', off);
    return () => {
      window.removeEventListener('online', on);
      window.removeEventListener('offline', off);
    };
  }, []);
  return online;
}

// Ensure PWA updates apply immediately when no code is actively running.
// Workbox serves precached assets first and updates the Service Worker in the
// background; listening for `controllerchange` reloads into the fresh bundle.
if (typeof window !== 'undefined' && 'serviceWorker' in navigator) {
  const hadController = Boolean(navigator.serviceWorker.controller);
  let refreshing = false;
  navigator.serviceWorker.addEventListener('controllerchange', () => {
    if (!hadController || refreshing) return;
    if (useCodeStore.getState().status === 'idle') {
      refreshing = true;
      window.location.reload();
    }
  });
  navigator.serviceWorker.getRegistration().then((reg) => {
    void reg?.update();
  });
}

const root = document.getElementById('root');
if (root) {
  createRoot(root).render(
    <StrictMode>
      <App />
    </StrictMode>,
  );
}
